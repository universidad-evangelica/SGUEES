CREATE PROCEDURE [dbo].[CLASS_SP_CONT_BALANCE_GENERAL_V1]
(
@ANIO AS INTEGER,
@MES AS INTEGER,
@CIERRE AS INTEGER,
@ACCION AS INTEGER
)
AS
BEGIN
IF @ACCION=0 BEGIN
	SELECT DISTINCT YEAR(TRAFVL) AS ANIO INTO #SELECT FROM TRANSAC
	ORDER BY YEAR(TRAFVL) DESC
	
	INSERT #SELECT (ANIO)
	SELECT DISTINCT YEAR(TRAFVL) AS ANIO FROM TRANSAC_HST
	ORDER BY YEAR(TRAFVL) DESC
	
	SELECT ANIO FROM #SELECT ORDER BY ANIO DESC
	
	DROP TABLE #SELECT
END
ELSE IF @ACCION=1 BEGIN
	SELECT CAST(NULL AS VARCHAR(5)) AS TRABNK, CAST(NULL AS VARCHAR(10)) AS TRABRN, CAST(NULL AS VARCHAR(5)) AS TRADCC,
			CAST(NULL AS MONEY) AS CARGO, CAST(NULL AS MONEY) AS ABONO, CAST(NULL AS VARCHAR(16)) AS TRAGLN,
			CAST(NULL AS INTEGER) AS TRAVDY, CAST(NULL AS INTEGER) AS TRAVDM INTO #TEMPMOV
	
	DELETE FROM #TEMPMOV
	
	IF @CIERRE=0 BEGIN
		INSERT INTO #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT TR.TRABNK, TR.TRABRN, TR.TRADCC, CASE WHEN TR.TRADCC='D' THEN TR.TRAAMT ELSE 0.00 END CARGO, 
			CASE WHEN TR.TRADCC='C' THEN TR.TRAAMT ELSE 0.00 END ABONO, 
			CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12))+REPLICATE('0',12-LEN(CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12)))) AS TRAGLN, 
			TR.TRAVDY, TR.TRAVDM
		FROM TRANSAC AS TR
		WHERE TR.TRATYP='R' AND LEFT(CAST(TR.TRAGLN AS VARCHAR(25)),1) IN ('1', '2', '3', '4', '5') 
		AND TR.TRAMOD <> 'P/G' AND TRALOT NOT IN ('2077') 
		AND YEAR(TR.TRAFVL)<@ANIO 
		AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))<>'9999'+CAST(@ANIO AS VARCHAR(5))
	    AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))<>'2078'+CAST(@ANIO AS VARCHAR(5))
	    AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))+TRADSC<>'2080'+CAST(@ANIO AS VARCHAR(5))+'Liquidación Cuenta'
		
		
		
		INSERT INTO #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT TR.TRABNK, TR.TRABRN, TR.TRADCC, CASE WHEN TR.TRADCC='D' THEN TR.TRAAMT ELSE 0.00 END CARGO, 
			CASE WHEN TR.TRADCC='C' THEN TR.TRAAMT ELSE 0.00 END ABONO, 
			CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12))+REPLICATE('0',12-LEN(CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12)))) AS TRAGLN, 			
			TR.TRAVDY, TR.TRAVDM --INTO #TEMPMOV
		FROM TRANSAC AS TR
		WHERE TR.TRATYP='R' AND LEFT(CAST(TR.TRAGLN AS VARCHAR(25)),1) IN ('1', '2', '3', '4', '5') 
		AND TR.TRAMOD <> 'P/G' AND TRALOT NOT IN ('2077') 
		AND YEAR(TR.TRAFVL)=@ANIO AND MONTH(TR.TRAFVL)<=@MES 
		AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))<>'9999'+CAST(@ANIO AS VARCHAR(5))
	    AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))<>'2078'+CAST(@ANIO AS VARCHAR(5))
	    AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))+TRADSC<>'2080'+CAST(@ANIO AS VARCHAR(5))+'Liquidación Cuenta'
		
		
		INSERT INTO #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT TR.TRABNK, TR.TRABRN, TR.TRADCC, CASE WHEN TR.TRADCC='D' THEN TR.TRAAMT ELSE 0.00 END CARGO, 
			CASE WHEN TR.TRADCC='C' THEN TR.TRAAMT ELSE 0.00 END ABONO, 
			CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12))+REPLICATE('0',12-LEN(CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12)))) AS TRAGLN, 
			TR.TRAVDY, TR.TRAVDM
		FROM TRANSAC_HST AS TR
		WHERE TR.TRATYP='R' AND LEFT(CAST(TR.TRAGLN AS VARCHAR(25)),1) IN ('1', '2', '3', '4', '5') 
		AND TR.TRAMOD <> 'P/G' AND TRALOT NOT IN ('2077') 
		AND YEAR(TR.TRAFVL)<@ANIO 
		AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))<>'9999'+CAST(@ANIO AS VARCHAR(5))
	    AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))<>'2078'+CAST(@ANIO AS VARCHAR(5))
	    AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))+TRADSC<>'2080'+CAST(@ANIO AS VARCHAR(5))+'Liquidación Cuenta'
		
		
		INSERT INTO #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT TR.TRABNK, TR.TRABRN, TR.TRADCC, CASE WHEN TR.TRADCC='D' THEN TR.TRAAMT ELSE 0.00 END CARGO, 
			CASE WHEN TR.TRADCC='C' THEN TR.TRAAMT ELSE 0.00 END ABONO, 
			CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12))+REPLICATE('0',12-LEN(CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12)))) AS TRAGLN, 
			TR.TRAVDY, TR.TRAVDM --INTO #TEMPMOV
		FROM TRANSAC_HST AS TR
		WHERE TR.TRATYP='R' AND LEFT(CAST(TR.TRAGLN AS VARCHAR(25)),1) IN ('1', '2', '3', '4', '5') 
		AND TR.TRAMOD <> 'P/G' AND TRALOT NOT IN ('2077') 
		AND YEAR(TR.TRAFVL)=@ANIO AND MONTH(TR.TRAFVL)<=@MES 
		AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))<>'9999'+CAST(@ANIO AS VARCHAR(5))
	    AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))<>'2078'+CAST(@ANIO AS VARCHAR(5))
	    AND CAST(TRALOT AS VARCHAR(5))+CAST(YEAR(TRAFVL) AS VARCHAR(5))+TRADSC<>'2080'+CAST(@ANIO AS VARCHAR(5))+'Liquidación Cuenta'
	END
	ELSE IF @CIERRE=1 BEGIN
		INSERT INTO #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT TR.TRABNK, TR.TRABRN, TR.TRADCC, CASE WHEN TR.TRADCC='D' THEN TR.TRAAMT ELSE 0.00 END CARGO, 
			CASE WHEN TR.TRADCC='C' THEN TR.TRAAMT ELSE 0.00 END ABONO, 
			CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12))+REPLICATE('0',12-LEN(CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12)))) AS TRAGLN, 
			TR.TRAVDY, TR.TRAVDM
		FROM TRANSAC AS TR
		WHERE TR.TRATYP='R' AND LEFT(CAST(TR.TRAGLN AS VARCHAR(25)),1) IN ('1', '2', '3', '4', '5') 
		AND TR.TRAMOD <> 'P/G' AND TRALOT NOT IN ('2077') 
		AND YEAR(TR.TRAFVL)<@ANIO 
		--AND TRALOT<>'9999'
		
		INSERT INTO #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT TR.TRABNK, TR.TRABRN, TR.TRADCC, CASE WHEN TR.TRADCC='D' THEN TR.TRAAMT ELSE 0.00 END CARGO, 
			CASE WHEN TR.TRADCC='C' THEN TR.TRAAMT ELSE 0.00 END ABONO, 
			CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12))+REPLICATE('0',12-LEN(CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12)))) AS TRAGLN, 
			TR.TRAVDY, TR.TRAVDM --INTO #TEMPMOV
		FROM TRANSAC AS TR
		WHERE TR.TRATYP='R' AND LEFT(CAST(TR.TRAGLN AS VARCHAR(25)),1) IN ('1', '2', '3', '4', '5') 
		AND TR.TRAMOD <> 'P/G' AND TRALOT NOT IN ('2077') 
		AND YEAR(TR.TRAFVL)=@ANIO AND MONTH(TR.TRAFVL)<=@MES 
		--AND TRALOT<>'9999'
		
		INSERT INTO #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT TR.TRABNK, TR.TRABRN, TR.TRADCC, CASE WHEN TR.TRADCC='D' THEN TR.TRAAMT ELSE 0.00 END CARGO, 
			CASE WHEN TR.TRADCC='C' THEN TR.TRAAMT ELSE 0.00 END ABONO, 
			CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12))+REPLICATE('0',12-LEN(CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12)))) AS TRAGLN, 
			TR.TRAVDY, TR.TRAVDM
		FROM TRANSAC_HST AS TR
		WHERE TR.TRATYP='R' AND LEFT(CAST(TR.TRAGLN AS VARCHAR(25)),1) IN ('1', '2', '3', '4', '5') 
		AND TR.TRAMOD <> 'P/G' AND TRALOT NOT IN ('2077') 
		AND YEAR(TR.TRAFVL)<@ANIO 
		--AND TRALOT<>'9999'
		
		INSERT INTO #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT TR.TRABNK, TR.TRABRN, TR.TRADCC, CASE WHEN TR.TRADCC='D' THEN TR.TRAAMT ELSE 0.00 END CARGO, 
			CASE WHEN TR.TRADCC='C' THEN TR.TRAAMT ELSE 0.00 END ABONO, 
			CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12))+REPLICATE('0',12-LEN(CAST(LEFT(TR.TRAGLN, 4) AS VARCHAR(12)))) AS TRAGLN, 
			TR.TRAVDY, TR.TRAVDM --INTO #TEMPMOV
		FROM TRANSAC_HST AS TR
		WHERE TR.TRATYP='R' AND LEFT(CAST(TR.TRAGLN AS VARCHAR(25)),1) IN ('1', '2', '3', '4', '5') 
		AND TR.TRAMOD <> 'P/G' AND TRALOT NOT IN ('2077') 
		AND YEAR(TR.TRAFVL)=@ANIO AND MONTH(TR.TRAFVL)<=@MES 
		--AND TRALOT<>'9999'
	END
	
	DECLARE @GASTOINGRESO AS MONEY
	
	SET @GASTOINGRESO = ISNULL((SELECT SUM(M.ABONO) - SUM(M.CARGO) FROM #TEMPMOV M WHERE LEFT(M.TRAGLN,1) IN ('5')),0.00)
						-
						ISNULL((SELECT SUM(M.CARGO) - SUM(M.ABONO) FROM #TEMPMOV M WHERE LEFT(M.TRAGLN,1) IN ('4')),0.00)
	
	IF @GASTOINGRESO > 0 BEGIN
		INSERT INTO  #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT '01', '001', 'C', 0.00, @GASTOINGRESO, '310600000000', @ANIO, @MES
	END
	ELSE IF @GASTOINGRESO < 0 BEGIN
		--SET @GASTOINGRESO=@GASTOINGRESO*-1
		INSERT INTO  #TEMPMOV (TRABNK, TRABRN, TRADCC, CARGO, ABONO, TRAGLN, TRAVDY, TRAVDM)
		SELECT '01', '001', 'C', @GASTOINGRESO, 0.00,  '310800000000', @ANIO, @MES
	END
	
	DECLARE @FECHA AS DATETIME
	SET @FECHA = CAST((DATEADD(s,-1,DATEADD(mm, DATEDIFF(m,0,'01/'+CAST(@MES AS VARCHAR(2))+'/'+CAST(@ANIO AS VARCHAR(5)))+1,0))) AS DATETIME)
	
	SELECT TRABNK, TRABRN, 	TRAGLN, CONDSC,
		CASE WHEN LEFT(TRAGLN,1) IN ('1','4') THEN SUM(ISNULL(CARGO,0.00))-SUM(ISNULL(ABONO,0.00)) ELSE SUM(ISNULL(ABONO,0.00))-SUM(ISNULL(CARGO,0.00)) END SALDO,
		LEFT(TRAGLN,2) AS TIPO_CUENTA, 
		CONVERT(CHAR(10), @FECHA, 103) AS FECHA
	FROM #TEMPMOV M, CONCAT
	WHERE TRABNK=CONBNK AND TRABRN=CONSUC AND TRAGLN=CONGLN AND TRABNK='01' AND TRABRN='001'
	GROUP BY TRABNK, TRABRN, TRAGLN, CONDSC
	ORDER BY TRAGLN ASC
	
	DROP TABLE #TEMPMOV
END
END
